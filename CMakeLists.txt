cmake_minimum_required(VERSION 3.20)
project(NgpCraftPackager VERSION 0.4.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This version targets Windows")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MINGW)
  add_link_options(-static -static-libgcc -static-libstdc++ -municode)
  add_link_options($<$<CONFIG:Release>:-s>)
  add_compile_options(-Wall -Wextra)
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(INPUT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/firmware/bios_hle.bin")
set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/bios_hle_data.hpp")
include(vendor/cmake/embed_bios.cmake)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS vendor/core/src/*.cpp)
add_library(core STATIC ${CORE_SOURCES} src/native_core.cpp)
target_include_directories(core PUBLIC vendor/core/include src PRIVATE vendor/core/src "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(src/player.rc.in "${CMAKE_CURRENT_BINARY_DIR}/player.rc" @ONLY)
add_executable(NgpCraftPlayer WIN32 src/player.cpp src/options.rc src/licenses.rc "${CMAKE_CURRENT_BINARY_DIR}/player.rc")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/player.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
target_link_libraries(NgpCraftPlayer PRIVATE core winmm gdi32 shell32 ole32 bcrypt)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc.in" CONTENT "201 RCDATA \"$<TARGET_FILE:NgpCraftPlayer>\"\n1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest\"\n")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc"
  DEPENDS NgpCraftPlayer "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc.in" src/app.manifest
  VERBATIM)
add_executable(NgpCraftPackager WIN32 src/packager.cpp src/options.rc src/licenses.rc "${CMAKE_CURRENT_BINARY_DIR}/packager-$<CONFIG>.rc")
add_dependencies(NgpCraftPackager NgpCraftPlayer)
target_link_libraries(NgpCraftPackager PRIVATE comdlg32 gdi32 shell32)

include(CTest)
if(BUILD_TESTING)
  add_executable(native_core_test tests/native_core_test.cpp)
  target_link_libraries(native_core_test PRIVATE core)
  add_test(NAME native_core COMMAND native_core_test)
  add_executable(video_surface_test tests/video_surface_test.cpp)
  target_include_directories(video_surface_test PRIVATE src)
  target_link_libraries(video_surface_test PRIVATE gdi32)
  add_test(NAME video_surface COMMAND video_surface_test)
endif()
